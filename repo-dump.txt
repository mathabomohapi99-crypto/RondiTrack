=== web/src/api/apiError.ts
// The one error type the whole app uses for failed API calls.
// problem = the problem+json body (read defensively, no hand-written API types).
export class ApiError extends Error {
  readonly status: number;
  readonly problem: Record<string, unknown> | null;
  readonly retryAfter: string | null; // raw Retry-After header text

  constructor(
    status: number,
    problem: Record<string, unknown> | null,
    retryAfter: string | null,
  ) {
    super(
      typeof problem?.title === 'string' ? problem.title : `Request failed with ${status}`,
    );
    this.name = 'ApiError';
    this.status = status;
    this.problem = problem;
    this.retryAfter = retryAfter;
  }
}

// Turns a failed Response into an ApiError (body may be empty, so we guard it)
export async function apiErrorFromResponse(res: Response): Promise<ApiError> {
  let problem: Record<string, unknown> | null = null;
  try {
    const body: unknown = await res.json();
    if (body && typeof body === 'object') problem = body as Record<string, unknown>;
  } catch {
    // no JSON body, that is fine
  }
  return new ApiError(res.status, problem, res.headers.get('Retry-After'));
}

import { API_BASE_URL } from './config';

// Returns the data or throws an ApiError (status + problem body + Retry-After)
export async function unwrap<T>(result: {
  data?: T;
  error?: unknown;
  response: Response;
}): Promise<T> {
  if (result.response.ok && result.data !== undefined) return result.data;
  const problem =
    result.error && typeof result.error === 'object'
      ? (result.error as Record<string, unknown>)
      : null;
  throw new ApiError(result.response.status, problem, result.response.headers.get('Retry-After'));
}

=== web/src/auth/session.ts
/*
 * WHY THE REFRESH MUST BE SINGLE-FLIGHT FOR MY API:
 * My API rotates the refresh cookie, so every refresh uses up the cookie it was sent with.
 * If five requests expire together and each sends its own refresh with the SAME old cookie,
 * the first wins and the other four are rejected ([FILL IN: the status you saw in check 5,
 * e.g. 401 or 409]), which would sign the member out. One shared refresh avoids that.
 *
 * This is the ONLY file in the project that reads or writes the access token.
 */
import { apiErrorFromResponse } from '../api/apiError';
import { API_BASE_URL } from '../api/config';

// Token lives only in this variable (never in browser storage, JS-set cookies, or React state)
let accessToken: string | null = null;
// The refresh that is currently running (shared by everyone who asks at the same time)
let refreshPromise: Promise<string | null> | null = null;
const signedOutListeners = new Set<() => void>();

// The rest of the app subscribes here to hear "member signed out"
export function onSignedOut(listener: () => void): () => void {
  signedOutListeners.add(listener);
  return () => {
    signedOutListeners.delete(listener);
  };
}

function notifySignedOut() {
  signedOutListeners.forEach((l) => l());
}

// Reads accessToken out of a JSON body without hand-writing a type
function tokenFrom(body: unknown): string | null {
  if (body && typeof body === 'object' && 'accessToken' in body && typeof body.accessToken === 'string') {
    return body.accessToken;
  }
  return null;
}

export async function login(email: string, password: string): Promise<void> {
  const res = await fetch(`${API_BASE_URL}/api/v1/auth/login`, {
    method: 'POST',
    credentials: 'include', // so the browser stores the refresh cookie
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email, password }),
  });
  if (!res.ok) throw await apiErrorFromResponse(res);
  const token = tokenFrom(await res.json());
  if (!token) throw new Error('Login response had no access token');
  accessToken = token;
}

export async function logout(): Promise<void> {
  const tokenBefore = accessToken;
  accessToken = null; // forget the token first, no matter what happens next
  try {
    await fetch(`${API_BASE_URL}/api/v1/auth/logout`, {
      method: 'POST',
      credentials: 'include',
      headers: tokenBefore ? { Authorization: `Bearer ${tokenBefore}` } : {},
    });
  } catch {
    // network failed: we still sign out locally
  } finally {
    notifySignedOut(); // always tell the app
  }
}

// The real refresh call. Success stores the token; failure clears it and signs out.
async function doRefresh(): Promise<string | null> {
  try {
    const res = await fetch(`${API_BASE_URL}/api/v1/auth/refresh`, {
      method: 'POST',
      credentials: 'include',
    });
    const token = res.ok ? tokenFrom(await res.json()) : null;
    if (!token) throw new Error('refresh failed');
    accessToken = token;
    return token;
  } catch {
    accessToken = null;
    notifySignedOut();
    return null;
  }
}

export function refreshAccessToken(): Promise<string | null> {
  // SINGLE-FLIGHT: if a refresh is already running, everyone gets that same promise.
  // .finally clears it (success OR failure) so the next expiry can start a new refresh.
  refreshPromise ??= doRefresh().finally(() => {
    refreshPromise = null;
  });
  return refreshPromise;
}

// Sends a copy of the request with the token and cookie settings added
function send(req: Request, token: string | null): Promise<Response> {
  const headers = new Headers(req.headers);
  if (token) headers.set('Authorization', `Bearer ${token}`);
  return fetch(new Request(req, { headers, credentials: 'include' }));
}

// The fetch we hand to openapi-fetch
export async function authFetch(request: Request): Promise<Response> {
  // A body can only be read once, so keep a clone for a possible retry
  const retryCopy = request.clone();

  // After a page reload memory is empty: get a token using the cookie first
  if (!accessToken) await refreshAccessToken();

  const tokenUsed = accessToken;
  const response = await send(request, tokenUsed);

  // Not a 401, or we had NO token to begin with: return now (this stops the endless loop)
  if (response.status !== 401 || !tokenUsed) return response;

  // If another request already got a newer token, use it instead of refreshing again
  const freshToken =
    accessToken && accessToken !== tokenUsed ? accessToken : await refreshAccessToken();

  if (!freshToken) return response; // refresh failed, doRefresh already signed us out
  return send(retryCopy, freshToken); // retry exactly once
}

// ONLY for tests: wipes module state between tests
export function __resetSessionForTests() {
  accessToken = null;
  refreshPromise = null;
  signedOutListeners.clear();
}

// NEW: "Due today / Due tomorrow / Due in N days / Overdue by N days".
// It re-renders ONCE just after each Johannesburg midnight, not every second (it only shows days).
import { useEffect, useState } from 'react'
import { dueCountdown, msUntilNextSaMidnight } from '@/lib/countdown'

export function DueCountdown({ due, cycleStatus }: { due: string; cycleStatus: string }) {
  const [now, setNow] = useState(() => new Date())

  useEffect(() => {
    if (cycleStatus !== 'Open') return // Closed / PaidOut cycles have no countdown, so no timer
    // setTimeout (not setInterval): one timer for the next SA midnight. +50ms lands just after it.
    const id = setTimeout(() => setNow(new Date()), msUntilNextSaMidnight(now) + 50)
    return () => clearTimeout(id) // cleanup: StrictMode and unmounting must not leave a timer behind
    // The effect READS `now`, so it must run again when `now` changes: that schedules the next midnight.
  }, [now, cycleStatus])

  const text = dueCountdown(due, now, cycleStatus)
  if (text === null) return null
  return <p className="text-sm font-medium">{text}</p>
}

=== web/src/components/ErrorView.tsx
import { useEffect, useState } from 'react';
import { describeError, rateLimitMessage } from '../lib/errorMessages';

// Counts down from "start" seconds; its timer is cleared when the component unmounts
function useCountdown(start: number | null): number | null {
  const [endsAt] = useState(() => (start === null ? null : Date.now() + start * 1000));
  const [now, setNow] = useState(() => Date.now());
  useEffect(() => {
    if (endsAt === null) return;
    const id = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(id);
  }, [endsAt]);
  return endsAt === null ? null : Math.max(0, Math.ceil((endsAt - now) / 1000));
}

=== web/src/lib/countdown.ts
// NEW: South African due-date maths. Pure functions: "now" is always passed in, never read here.
const MS_PER_DAY = 86_400_000
// South Africa is UTC+2 all year with no daylight saving since 1944, so a fixed offset is safe.
const SA_OFFSET_MS = 2 * 3_600_000
const CALENDAR_DATE = /^\d{4}-\d{2}-\d{2}$/

// Which South African calendar day an instant falls on (a whole number of days since 1970).
export function saDayIndex(instant: Date): number {
  return Math.floor((instant.getTime() + SA_OFFSET_MS) / MS_PER_DAY)
}

// The South African calendar day a due date means. "2026-10-31" is that day itself;
// an instant like "2026-10-31T21:59:59Z" is the SA day it falls on.
function dueDayIndex(due: string): number {
  if (CALENDAR_DATE.test(due)) {
    const year = Number(due.slice(0, 4))
    const month = Number(due.slice(5, 7))
    const day = Number(due.slice(8, 10))
    return Date.UTC(year, month - 1, day) / MS_PER_DAY
  }
  const instant = new Date(due)
  if (Number.isNaN(instant.getTime())) throw new Error(`not a valid due date: ${due}`)
  return saDayIndex(instant)
}

// The moment the due day ENDS in Johannesburg. "2026-10-31" -> 2026-10-31T22:00:00Z (not 00:00Z).
export function dueDateToDeadline(due: string): Date {
  return new Date((dueDayIndex(due) + 1) * MS_PER_DAY - SA_OFFSET_MS)
}

// Milliseconds until the next Johannesburg midnight. Exactly at midnight it returns 24 hours, never 0.
export function msUntilNextSaMidnight(now: Date): number {
  return (saDayIndex(now) + 1) * MS_PER_DAY - SA_OFFSET_MS - now.getTime()
}

function days(n: number): string {
  return n === 1 ? 'day' : 'days'
}

// The countdown words, or null when no countdown should show (cycle Closed or PaidOut).
export function dueCountdown(due: string, now: Date, cycleStatus: string): string | null {
  if (cycleStatus !== 'Open') return null

  if (now.getTime() >= dueDateToDeadline(due).getTime()) {
    const late = saDayIndex(now) - dueDayIndex(due)
    return `Overdue by ${late} ${days(late)}`
  }

  const left = dueDayIndex(due) - saDayIndex(now)
  if (left === 0) return 'Due today'
  if (left === 1) return 'Due tomorrow'
  return `Due in ${left} days`
}

// NEW: formats dates in South African time, whatever time zone the laptop is set to.
const TIME_ZONE = 'Africa/Johannesburg'

// "01 Nov 2026, 00:30"
const dateTimeFormatter = new Intl.DateTimeFormat('en-ZA', {
  dateStyle: 'medium',
  timeStyle: 'short',
  timeZone: TIME_ZONE,
})

// "01 Nov 2026"
const dateFormatter = new Intl.DateTimeFormat('en-ZA', {
  dateStyle: 'medium',
  timeZone: TIME_ZONE,
})

// Matches a plain calendar date like "2026-10-31" (no time, no Z)
const CALENDAR_DATE = /^\d{4}-\d{2}-\d{2}$/

function toDate(value: string): Date {
  // A plain calendar date is already a South African day. Noon UTC (14:00 in SA)
  // means no time zone can push it onto the wrong day.
  const date = CALENDAR_DATE.test(value) ? new Date(`${value}T12:00:00Z`) : new Date(value)
  if (Number.isNaN(date.getTime())) {
    throw new Error(`not a valid date: ${value}`)
  }
  return date
}

// For UTC instants like "2026-10-31T22:30:00Z" (date AND time, in SA time)
export function formatDateTime(value: string): string {
  return dateTimeFormatter.format(toDate(value))
}

// Date only. Works with both a UTC instant and a calendar date
export function formatDate(value: string): string {
  return dateFormatter.format(toDate(value))
}

  fieldErrors: Record<string, string[]>; // keys are lower-case, e.g. "email"
};

// Retry-After is either seconds ("23") or an HTTP date. Returns seconds or null.
export function parseRetryAfter(header: string | null | undefined, now: number = Date.now()): number | null {
  if (!header) return null;
  const text = header.trim();
  if (/^\d+$/.test(text)) return Number(text);
  const when = Date.parse(text);
  if (Number.isNaN(when)) return null;
  return Math.max(0, Math.ceil((when - now) / 1000)); // date in the past = 0
}

export function rateLimitMessage(seconds: number | null): string {
  if (seconds === null) return 'Too many requests. Please wait a moment and try again.';
  return `Too many requests. Try again in ${seconds} ${seconds === 1 ? 'second' : 'seconds'}.`;
}

const asString = (v: unknown): string | null => (typeof v === 'string' && v ? v : null);

// Turns ANY error into plain words. Never uses the "detail" field (it can leak internals on a 500).
export function describeError(error: unknown, context: 'data' | 'login' = 'data'): DescribedError {
  const empty = { correlationId: null, retryAfterSeconds: null, fieldErrors: {} };

  // fetch itself failed (API down, no internet): there is no response and no reference
  if (error instanceof TypeError) {
    return {
      ...empty,
      kind: 'network',
      title: "Can't reach RondiTrack",
      message: 'Check your connection.',
    };
  }

  if (!(error instanceof ApiError)) {
    return {
      ...empty,
      kind: 'generic',
      title: 'Something went wrong',
      message: 'Something went wrong on our side. Please try again.',
    };
  }

  const correlationId = asString(error.problem?.correlationId);
  const base = { ...empty, correlationId };

  if (error.status === 429) {
    const seconds = parseRetryAfter(error.retryAfter);
    return {
      ...base,
      kind: 'rateLimited',
      title: 'Too many requests',
      message: rateLimitMessage(seconds),
      retryAfterSeconds: seconds,
    };
  }
  if (error.status === 401) {
    return context === 'login'
      ? { ...base, kind: 'badCredentials', title: 'Sign in failed', message: 'Email or password is incorrect.' }
      : { ...base, kind: 'sessionEnded', title: 'Session ended', message: 'Please sign in again.' };
  }
  if (error.status === 404) {
    return {
      ...base,
      kind: 'notFound',
      title: 'Stokvel not found',
      message: "This stokvel doesn't exist, or you are not a member of it.",
    };
  }
  if (error.status === 400) {
    const fieldErrors: Record<string, string[]> = {};
    const raw = error.problem?.errors;
    if (raw && typeof raw === 'object') {
      for (const [key, value] of Object.entries(raw)) {
        if (Array.isArray(value)) fieldErrors[key.toLowerCase()] = value.map(String);
      }
    }
    return { ...base, kind: 'validation', title: 'Please check your details', message: 'Some fields need your attention.', fieldErrors };
  }
  return {
    ...base,
    kind: 'generic',
    title: 'Something went wrong',
    message: 'Something went wrong on our side. Please try again in a moment.',
  };
}

=== web/src/lib/groupContributionsByCycle.ts
// NEW: groups contributions under their cycle. A pure function: no state, no effects, called during render.
// Generic on purpose: it only needs a cycle id + number and a contribution cycleId, so the real types come from the API.
export const UNKNOWN_CYCLE_KEY = 'unknown-cycle'

export type CycleGroup<C, T> = {
  key: string // the cycle id (or "unknown-cycle"): open/closed state is keyed by this, never by position
  cycle: C | null // null only for the "Unknown cycle" group
  contributions: T[]
}

export function groupContributionsByCycle<
  C extends { id: string; number: number },
  T extends { cycleId: string },
>(cycles: readonly C[], contributions: readonly T[]): CycleGroup<C, T>[] {
  const knownIds = new Set(cycles.map((cycle) => cycle.id))
  const byCycle = new Map<string, T[]>()
  const orphans: T[] = []

  // Keep each contribution in the order the API gave it (newest first)
  for (const contribution of contributions) {
    if (knownIds.has(contribution.cycleId)) {
      const list = byCycle.get(contribution.cycleId) ?? []
      list.push(contribution)
      byCycle.set(contribution.cycleId, list)
    } else {
      orphans.push(contribution) // never dropped: it goes to "Unknown cycle"
    }
  }

  // Newest cycle first (highest number). A cycle with no contributions still gets a group.
  const groups: CycleGroup<C, T>[] = [...cycles]
    .sort((a, b) => b.number - a.number)
    .map((cycle) => ({ key: cycle.id, cycle, contributions: byCycle.get(cycle.id) ?? [] }))

  if (orphans.length > 0) {
    groups.push({ key: UNKNOWN_CYCLE_KEY, cycle: null, contributions: orphans })
  }
  return groups
}

// NEW: turns an integer number of cents into "R 150,50" without ever using fractional floats.
const zarFormatter = new Intl.NumberFormat('en-ZA', {
  style: 'currency',
  currency: 'ZAR',
})

export function formatCents(cents: number): string {
  // Reject 10.5, NaN, Infinity, or numbers too big to be exact
  if (!Number.isSafeInteger(cents)) {
    throw new Error('not a whole number of cents')
  }

  // BigInt keeps everything as whole numbers, so there is no float rounding at all
  const big = BigInt(cents)
  const negative = big < 0n
  const abs = negative ? -big : big

  // Integer division and remainder: 15050 cents -> 150 rands and 50 cents
  const rands = abs / 100n
  const rest = abs % 100n

  // Build an exact decimal string like "150.50" (pad so 5 cents becomes "05")
  const decimal = `${negative ? '-' : ''}${rands}.${rest.toString().padStart(2, '0')}`

  // Intl accepts an exact decimal string, so it formats without turning it into a float
  return zarFormatter.format(decimal as `${number}`)
}

// Step 7 name: later files import formatRand, so it forwards to formatCents
export function formatRand(value: number): string {
  return formatCents(value)
}

// NEW: remembers the theme choice and puts the "dark" class on <html>. System follows the OS live.
import { useEffect, useState, type ReactNode } from 'react'
import { THEME_STORAGE_KEY, ThemeContext, type Theme } from './theme-context'

// Storage can be blocked (private browsing), so every read and write is guarded
function readStoredTheme(): Theme {
  try {
    const value = localStorage.getItem(THEME_STORAGE_KEY)
    if (value === 'light' || value === 'dark' || value === 'system') return value
  } catch {
    // storage blocked: fall back to the default
  }
  return 'system'
}

function applyTheme(theme: Theme) {
  const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches
  const dark = theme === 'dark' || (theme === 'system' && prefersDark)
  document.documentElement.classList.toggle('dark', dark)
  document.documentElement.style.colorScheme = dark ? 'dark' : 'light'
}

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [theme, setThemeState] = useState<Theme>(readStoredTheme)

  useEffect(() => {
    applyTheme(theme)
    if (theme !== 'system') return
    // System mode: re-apply whenever the operating system switches light/dark
    const media = window.matchMedia('(prefers-color-scheme: dark)')
    const onChange = () => applyTheme('system')
    media.addEventListener('change', onChange)
    return () => media.removeEventListener('change', onChange)
  }, [theme])

  function setTheme(next: Theme) {
    setThemeState(next)
    try {
      localStorage.setItem(THEME_STORAGE_KEY, next)
    } catch {
      // storage blocked: the choice still works until the page reloads
    }
  }

  return <ThemeContext.Provider value={{ theme, setTheme }}>{children}</ThemeContext.Provider>
}

    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status422UnprocessableEntity)]
    public async Task<ActionResult<ContributionResponse>> RecordContribution(

    {

        await access.EnsureAsync(User, stokvelId, Policies.RecordContribution, request.UserId);

            throw new DomainValidationException("The Idempotency-Key header is required.");

        if (replayed) Response.Headers[ApiHeaders.IdempotencyReplayed] = "true";

=== API/Services/ContributionServices.cs
using System.Security.Cryptography;
using System.Text;
using RondiTrack.Data;
using RondiTrack.Domain;
using RondiTrack.Dtos;
using RondiTrack.Telemetry;   // NEW 6.2

namespace RondiTrack.Services;

public interface IContributionService
{
    Task<(ContributionResponse Response, bool WasReplayed)> RecordContributionAsync(
        Guid stokvelId, ContributionRequest request, string idempotencyKey);
}

public sealed class ContributionService(
    IStokvelRepository stokvels,
    IUserRepository users,
    IContributionCycleRepository cycles,
    IContributionRepository contributions,
    IIdempotencyStore idempotencyStore) : IContributionService
{
    public async Task<(ContributionResponse Response, bool WasReplayed)> RecordContributionAsync(
        Guid stokvelId, ContributionRequest request, string idempotencyKey)
    {
        var requestHash = ComputeHash(stokvelId, request);

        var existing = await idempotencyStore.GetAsync(idempotencyKey);
        if (existing is not null)
        {
            if (existing.RequestHash != requestHash)
                throw new DomainConflictException("This Idempotency-Key was already used with a different request body.");

            return (existing.Response, true);
        }

        var stokvel = await stokvels.GetByIdAsync(stokvelId)
            ?? throw new DomainNotFoundException($"Stokvel {stokvelId} was not found.");

        var user = await users.GetByIdAsync(request.UserId)
            ?? throw new DomainReferenceException($"User {request.UserId} does not exist.");

        if (!stokvel.HasMember(user.Id))
            throw new DomainReferenceException("That user is not a member of this stokvel.");   // CHANGED 6.2: no name, no stokvel name

        var cycle = await cycles.GetByIdAsync(request.CycleId);
        if (cycle is null || cycle.StokvelId != stokvelId)
            throw new DomainReferenceException($"Cycle {request.CycleId} does not exist for this stokvel.");

        if (await contributions.ExistsAsync(stokvelId, user.Id, request.CycleId))
            throw new DomainConflictException("This member has already contributed for this cycle.");   // CHANGED 6.2: no name

        var contribution = new Contribution(stokvelId, user.Id, request.CycleId, request.Amount);
        await contributions.AddAsync(contribution);
        RondiMetrics.ContributionRecorded();   // NEW 6.2: counted only after the row is really saved

        var response = contribution.ToResponse();
        await idempotencyStore.SaveAsync(idempotencyKey, new IdempotencyRecord(requestHash, response));

        return (response, false);
    }

    private static string ComputeHash(Guid stokvelId, ContributionRequest request)
    {
        var raw = $"{stokvelId}|{request.UserId}|{request.CycleId}|{request.Amount}";
        var bytes = SHA256.HashData(Encoding.UTF8.GetBytes(raw));
        return Convert.ToHexString(bytes);
    }
}

using FluentValidation;
using Microsoft.AspNetCore.Diagnostics;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Npgsql;
using RondiTrack.Api.Common.Paging;
using RondiTrack.Auth;
using RondiTrack.Common;
using RondiTrack.Domain;
using RondiTrack.Telemetry;
using RondiTrack.Web;
using System.Text.Json;

namespace RondiTrack.Errors;

public sealed class RondiExceptionHandler(ILogger<RondiExceptionHandler> logger) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(HttpContext httpContext, Exception exception, CancellationToken cancellationToken)
    {
        // UseExceptionHandler wipes response headers (including CORS), so put them back
        await CorsSetup.ReapplyAsync(httpContext);

        var correlationId = CorrelationId.For(httpContext);

        var status = StatusCodes.Status500InternalServerError;
        var title = "Internal Server Error";
        var detail = "An unexpected error occurred.";
        Dictionary<string, string[]>? errors = null;   // field -> messages, validation only

        if (exception is ValidationException vex)
        {
            status = StatusCodes.Status400BadRequest;
            title = "Validation Failed";
            detail = "One or more fields are invalid.";
            errors = vex.Errors
                .GroupBy(e => JsonNamingPolicy.CamelCase.ConvertName(e.PropertyName))
                .ToDictionary(g => g.Key, g => g.Select(e => e.ErrorMessage).ToArray());
        }
        else if (exception is DomainNotFoundException) { status = 404; title = "Not Found"; detail = exception.Message; }
        else if (exception is DomainConflictException) { status = 409; title = "Conflict"; detail = exception.Message; }
        else if (exception is DomainReferenceException) { status = 422; title = "Unprocessable Entity"; detail = exception.Message; }
        else if (exception is DomainValidationException) { status = 400; title = "Bad Request"; detail = exception.Message; }
        else if (exception is ApiBadRequestException) { status = 400; title = "Bad Request"; detail = exception.Message; }
        else if (exception is PreconditionRequiredException) { status = 428; title = "Precondition Required"; detail = exception.Message; }
        else if (exception is InvalidCredentialsException or InvalidRefreshTokenException)
        {
            status = 401; title = "Unauthorized"; detail = exception.Message;
            httpContext.Response.Headers.WWWAuthenticate = "Bearer";
        }
        else if (exception is ForbiddenException) { status = 403; title = "Forbidden"; detail = exception.Message; }
        else if (exception is DbUpdateConcurrencyException)
        {
            if (httpContext.Request.Headers.ContainsKey("If-Match"))
            {
                status = 412; title = "Precondition Failed";
                detail = "This record was changed by someone else since you loaded it. Reload it and try again.";
            }
            else
            {
                status = 409; title = "Conflict";
                detail = "This record was changed by someone else at the same time. Reload it and try again.";
            }
        }
        else if (exception is DbUpdateException { InnerException: PostgresException { SqlState: PostgresErrorCodes.UniqueViolation } })
        {
            status = 409; title = "Conflict";
            detail = "That record already exists (it would break a uniqueness rule).";
        }

        if (status >= 500)
            logger.LogError(exception, "Request failed with status {Status}. CorrelationId: {CorrelationId}", status, correlationId);
        else
            logger.LogInformation("Request rejected with status {Status} ({ExceptionType}). CorrelationId: {CorrelationId}",
                status, exception.GetType().Name, correlationId);

        var problemDetails = new ProblemDetails { Status = status, Title = title, Detail = detail };
        problemDetails.Extensions["correlationId"] = correlationId;
        if (errors is not null) problemDetails.Extensions["errors"] = errors;

        httpContext.Response.StatusCode = status;
        httpContext.Response.ContentType = "application/problem+json";
        await httpContext.Response.WriteAsync(JsonSerializer.Serialize(problemDetails), cancellationToken);
        return true;
    }
}

=== API/Web/CorsSetup.cs
using Microsoft.AspNetCore.Cors.Infrastructure;

namespace RondiTrack.Web;

public static class CorsSetup
{
    public static IServiceCollection AddFrontendCors(this IServiceCollection services)
    {
        services.AddCors();
        // Options pattern = read lazily, so test UseSetting overrides are always seen
        services.AddOptions<CorsOptions>().Configure<IConfiguration>((o, config) =>
        {
            var origins = config.GetSection("Cors:AllowedOrigins").Get<string[]>() ?? [];
            if (origins.Any(x => x.Contains('*') || x.EndsWith('/')))
                throw new InvalidOperationException("Cors:AllowedOrigins must be exact origins: no '*', no trailing slash.");

            o.AddPolicy(CorsPolicies.Frontend, p => p
                .WithOrigins(origins)                                   // empty list = nobody allowed
                .WithMethods("GET", "POST", "PUT", "PATCH", "DELETE")
                .WithHeaders("Content-Type", "Authorization", ApiHeaders.IdempotencyKey, "If-Match", "If-None-Match")
                .WithExposedHeaders("Retry-After", ApiHeaders.IdempotencyReplayed, "ETag", "Location")
                .AllowCredentials()
                .SetPreflightMaxAge(TimeSpan.FromMinutes(10)));
        });
        return services;
    }

    /// The exception handler clears response headers, so CORS headers set earlier are lost on
    /// exception-based errors (404/403/409/500). Call this at the top of the handler to put them back.
    public static async Task ReapplyAsync(HttpContext ctx)
    {
        var sp = ctx.RequestServices;
        if (sp is null) return;                                   // bare DefaultHttpContext in unit tests: nothing to reapply
        var provider = sp.GetService<ICorsPolicyProvider>();
        var service = sp.GetService<ICorsService>();
        if (provider is null || service is null) return;          // CORS not registered
        var policy = await provider.GetPolicyAsync(ctx, CorsPolicies.Frontend);
        if (policy is null) return;
        service.ApplyResult(service.EvaluatePolicy(ctx, policy), ctx.Response);
    }
}

=== API/Web/RateLimitingSetup.cs
using System.Security.Claims;
using System.Threading.RateLimiting;
using Microsoft.AspNetCore.RateLimiting;
using RondiTrack.Telemetry;

namespace RondiTrack.Web;

public static class RateLimitingSetup
{
    public static IServiceCollection AddRondiRateLimiting(this IServiceCollection services)
    {
        services.AddRateLimiter(_ => { });
        services.AddOptions<RateLimiterOptions>().Configure<IConfiguration>((o, cfg) =>
        {
            o.RejectionStatusCode = StatusCodes.Status429TooManyRequests;   // default is 503
            o.OnRejected = WriteRejectionAsync;

            var login = cfg.GetSection("RateLimiting:Login");
            o.AddPolicy(RateLimitPolicies.Login, ctx =>
                RateLimitPartition.GetFixedWindowLimiter(ClientIp(ctx), _ => new FixedWindowRateLimiterOptions
                {
                    PermitLimit = login.GetValue("PermitLimit", 5),
                    Window = TimeSpan.FromSeconds(login.GetValue("WindowSeconds", 60.0)),
                    QueueLimit = 0,
                    AutoReplenishment = true
                }));

            o.AddPolicy(RateLimitPolicies.Contribution, ctx =>
                RateLimitPartition.GetTokenBucketLimiter(UserKey(ctx), _ => Bucket(cfg.GetSection("RateLimiting:Contribution"))));

            o.AddPolicy(RateLimitPolicies.Payout, ctx =>
                RateLimitPartition.GetTokenBucketLimiter(UserKey(ctx), _ => Bucket(cfg.GetSection("RateLimiting:Payout"))));
        });
        return services;
    }

    private static TokenBucketRateLimiterOptions Bucket(IConfigurationSection s) => new()
    {
        TokenLimit = s.GetValue("TokenLimit", 30),
        TokensPerPeriod = s.GetValue("TokensPerPeriod", 5),
        ReplenishmentPeriod = TimeSpan.FromSeconds(s.GetValue("ReplenishSeconds", 60.0)),
        QueueLimit = 0,              // reject, never hold a money request open
        AutoReplenishment = true
    };

    private static string ClientIp(HttpContext c) => "ip:" + (c.Connection.RemoteIpAddress?.ToString() ?? "unknown");

    // signed-in user id; falls back to IP only for unauthenticated callers (they'll get 401 afterwards)
    private static string UserKey(HttpContext c) =>
        c.User.FindFirstValue("sub") is { Length: > 0 } sub ? "user:" + sub : ClientIp(c);

    private static async ValueTask WriteRejectionAsync(OnRejectedContext ctx, CancellationToken ct)
    {
        var http = ctx.HttpContext;
        var seconds = 1;
        if (ctx.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retry))
            seconds = Math.Max(1, (int)Math.Ceiling(retry.TotalSeconds));

        http.Response.StatusCode = StatusCodes.Status429TooManyRequests;
        http.Response.Headers.RetryAfter = seconds.ToString();

        var problem = new Microsoft.AspNetCore.Mvc.ProblemDetails
        {
            Status = 429,
            Title = "Too many requests",
            Detail = $"Rate limit reached. Try again in {seconds} second(s).",
            Type = "https://www.rfc-editor.org/rfc/rfc6585#section-4"
        };
        problem.Extensions["correlationId"] = CorrelationId.For(http);   // your 6.2 helper
        await http.Response.WriteAsJsonAsync(problem, options: null, contentType: "application/problem+json", ct);
    }
}
