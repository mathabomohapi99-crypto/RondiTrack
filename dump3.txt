[Route($"{ApiRoutes.V1}/stokvels")]
[ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status401Unauthorized)]
[ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status403Forbidden)]
public class CycleContributionsController(
    IContributionService contributionService,
    IStokvelAccess access) : RondiControllerBase
{
    /// <summary>
    /// Records a contribution in one cycle. Amount is integer cents. A member may record their own;
    /// a Treasurer or Admin may record for anyone. Requires an Idempotency-Key header.
    /// </summary>
    [HttpPost("{stokvelId:guid}/cycles/{cycleId:guid}/contributions")]
    [EnableRateLimiting(RateLimitPolicies.Contribution)]
    [RequireStokvelMember]
    [ProducesResponseType(typeof(ContributionResponse), StatusCodes.Status201Created)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status400BadRequest)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status404NotFound)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status409Conflict)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status422UnprocessableEntity)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status429TooManyRequests)]
    public async Task<ActionResult<ContributionResponse>> RecordInCycle(
        Guid stokvelId,
        Guid cycleId,
        RecordContributionRequest request,
        [FromHeader(Name = "Idempotency-Key")] string? idempotencyKey)
    {
        // Own contribution = ok, someone else's = officer only (403 otherwise). Same rule as the old endpoint.
        await access.EnsureAsync(User, stokvelId, Policies.RecordContribution, request.UserId);

        if (string.IsNullOrWhiteSpace(idempotencyKey))
            throw new DomainValidationException("The Idempotency-Key header is required.");

        // cents -> decimal is exact (decimal division), so no rounding happens here.
        var amount = request.AmountInCents / 100m;

        var (response, replayed) = await contributionService.RecordContributionAsync(
            stokvelId, new ContributionRequest(request.UserId, cycleId, amount), idempotencyKey);

        if (replayed) Response.Headers[ApiHeaders.IdempotencyReplayed] = "true";
        return StatusCode(StatusCodes.Status201Created, response);
    }
}

using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

namespace RondiTrack.Api.Common.Paging;

// CHANGED: LastId is now a Guid (your ids are Guids, not ints).
// Context = fingerprint of the sort + filters, so we can detect a token reused with different ones.
public record PageCursor(string Context, string LastValue, Guid LastId);

public static class PageTokenCodec
{
    // Turns the cursor into an opaque base64url string the client cannot read.
    public static string Encode(PageCursor cursor)
    {
        var json = JsonSerializer.SerializeToUtf8Bytes(cursor);
        return Convert.ToBase64String(json).Replace('+', '-').Replace('/', '_').TrimEnd('=');
    }

    // Reads a token back. A broken token, or a token from a different sort/filter, gives a 400.
    public static PageCursor Decode(string token, string expectedContext)
    {
        PageCursor? cursor;
        try
        {
            var b64 = token.Replace('-', '+').Replace('_', '/');
            b64 = b64.PadRight(b64.Length + (4 - b64.Length % 4) % 4, '=');
            cursor = JsonSerializer.Deserialize<PageCursor>(Convert.FromBase64String(b64));
        }
        catch
        {
            throw new ApiBadRequestException("pageToken is invalid.");
        }

        if (cursor is null)
            throw new ApiBadRequestException("pageToken is invalid.");

        if (cursor.Context != expectedContext)
            throw new ApiBadRequestException(
                "pageToken was issued for a different sort or filter. Start again without a pageToken.");

        return cursor;
    }

    // Short fingerprint of sort + filters (same inputs always give the same text).
    public static string ContextOf(params object?[] parts)
    {
        var text = string.Join("|", parts.Select(p => p?.ToString() ?? ""));
        return Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(text)))[..12];
    }
}

using Microsoft.EntityFrameworkCore;
using RondiTrack.Api.Common.Paging;

// CHANGED 7.2: the rows now carry CycleId, integer cents and Status, and no email.
public sealed class ContributionPagingService(RondiTrackDbContext db)
{
    private static readonly string[] AllowedSorts = ["date", "-date", "amount", "-amount"];

    public async Task<PagedResponse<ContributionDetailResponse>> ListAsync(
        Guid cycleId, ContributionListQuery q, CancellationToken ct)
    {
        var pageSize = PagingDefaults.Resolve(q.PageSize);

        var sort = string.IsNullOrWhiteSpace(q.Sort) ? "date" : q.Sort.Trim().ToLowerInvariant();
        if (!AllowedSorts.Contains(sort))
            throw new ApiBadRequestException(
                $"Unknown sort '{q.Sort}'. Allowed: {string.Join(", ", AllowedSorts)}.");

        var from = ToUtc(q.DateFrom);
        var to = ToUtc(q.DateTo);
        if (from.HasValue && to.HasValue && from > to)
            throw new ApiBadRequestException("dateFrom cannot be after dateTo.");

        IQueryable<Contribution> query = db.Contributions
            .AsNoTracking()
            .Where(c => c.CycleId == cycleId);

        if (q.UserId.HasValue)
        {
            var uid = q.UserId.Value;
            query = query.Where(c => c.UserId == uid);
        }
        if (from.HasValue)
        {
            var f = from.Value;
            query = query.Where(c => c.RecordedAtUtc >= f);
        }
        if (to.HasValue)
        {
            var t = to.Value;
            query = query.Where(c => c.RecordedAtUtc <= t);
        }

        var context = PageTokenCodec.ContextOf(
            sort, cycleId, q.UserId, from?.ToString("O"), to?.ToString("O"));

        if (!string.IsNullOrEmpty(q.PageToken))
        {
            var cursor = PageTokenCodec.Decode(q.PageToken, context);
            query = ApplyAfterCursor(query, sort, cursor);
        }

        query = ApplyOrder(query, sort);

        // Ask for ONE extra row. If it comes back, there is a next page.
        var rows = await query
            .Take(pageSize + 1)
            .Select(c => new
            {
                c.Id, c.CycleId, c.UserId,
                MemberName = c.Member.User.FullName,
                c.Member.Role, c.Member.RotationPosition,
                c.Amount, c.Status, c.RecordedAtUtc
            })
            .ToListAsync(ct);

        var hasMore = rows.Count > pageSize;
        var pageRows = rows.Take(pageSize).ToList();

        var next = "";
        if (hasMore)
        {
            var last = pageRows[^1];
            var lastValue = sort.EndsWith("date")
                ? last.RecordedAtUtc.ToString("O", CultureInfo.InvariantCulture)
                : last.Amount.ToString(CultureInfo.InvariantCulture);
            next = PageTokenCodec.Encode(new PageCursor(context, lastValue, last.Id));
        }

        // money becomes integer cents here
        var page = pageRows.Select(r => new ContributionDetailResponse(
            r.Id, r.CycleId, r.UserId, r.MemberName, r.Role, r.RotationPosition,
            Money.ToCents(r.Amount), r.Status, r.RecordedAtUtc)).ToList();

        return new PagedResponse<ContributionDetailResponse>(page, next);
    }

    private static IQueryable<Contribution> ApplyOrder(IQueryable<Contribution> q, string sort) => sort switch
    {
        "date"   => q.OrderBy(c => c.RecordedAtUtc).ThenBy(c => c.Id),
        "-date"  => q.OrderByDescending(c => c.RecordedAtUtc).ThenByDescending(c => c.Id),
        "amount" => q.OrderBy(c => c.Amount).ThenBy(c => c.Id),
        _        => q.OrderByDescending(c => c.Amount).ThenByDescending(c => c.Id),
    };

    private static IQueryable<Contribution> ApplyAfterCursor(
        IQueryable<Contribution> q, string sort, PageCursor cur)
    {
        var lastId = cur.LastId;

        if (sort.EndsWith("date"))
        {
            if (!DateTime.TryParse(cur.LastValue, CultureInfo.InvariantCulture,
                    DateTimeStyles.RoundtripKind, out var parsed))
                throw new ApiBadRequestException("pageToken is invalid.");

            var d = DateTime.SpecifyKind(parsed, DateTimeKind.Utc);
            return sort == "date"
                ? q.Where(c => c.RecordedAtUtc > d || (c.RecordedAtUtc == d && c.Id.CompareTo(lastId) > 0))
                : q.Where(c => c.RecordedAtUtc < d || (c.RecordedAtUtc == d && c.Id.CompareTo(lastId) < 0));
        }

        if (!decimal.TryParse(cur.LastValue, NumberStyles.Number, CultureInfo.InvariantCulture, out var a))
            throw new ApiBadRequestException("pageToken is invalid.");

        return sort == "amount"
            ? q.Where(c => c.Amount > a || (c.Amount == a && c.Id.CompareTo(lastId) > 0))
            : q.Where(c => c.Amount < a || (c.Amount == a && c.Id.CompareTo(lastId) < 0));
    }

    private static DateTime? ToUtc(DateTime? value) => value is null ? null
        : value.Value.Kind == DateTimeKind.Utc ? value
        : value.Value.Kind == DateTimeKind.Local ? value.Value.ToUniversalTime()
        : DateTime.SpecifyKind(value.Value, DateTimeKind.Utc);
}

public sealed class ContributionService(
    IStokvelRepository stokvels,
    IUserRepository users,
    IContributionCycleRepository cycles,
    IContributionRepository contributions,
    IIdempotencyStore idempotencyStore) : IContributionService
{
    public async Task<(ContributionResponse Response, bool WasReplayed)> RecordContributionAsync(
        Guid stokvelId, ContributionRequest request, string idempotencyKey)
    {
        var requestHash = ComputeHash(stokvelId, request);

        var existing = await idempotencyStore.GetAsync(idempotencyKey);
        if (existing is not null)
        {
            if (existing.RequestHash != requestHash)
                throw new DomainConflictException("This Idempotency-Key was already used with a different request body.");

            return (existing.Response, true);
        }

        var stokvel = await stokvels.GetByIdAsync(stokvelId)
            ?? throw new DomainNotFoundException($"Stokvel {stokvelId} was not found.");

        var user = await users.GetByIdAsync(request.UserId)
            ?? throw new DomainReferenceException($"User {request.UserId} does not exist.");

        if (!stokvel.HasMember(user.Id))
            throw new DomainReferenceException("That user is not a member of this stokvel.");   // CHANGED 6.2: no name, no stokvel name

        var cycle = await cycles.GetByIdAsync(request.CycleId);
        if (cycle is null || cycle.StokvelId != stokvelId)
            throw new DomainReferenceException($"Cycle {request.CycleId} does not exist for this stokvel.");

        // NEW 7.4: a Closed or PaidOut cycle takes no more money. The form shows this detail if the cycle closes mid-edit.
        if (cycle.Status != CycleStatus.Open)
            throw new DomainConflictException("This cycle is no longer open, so contributions cannot be recorded.");

        if (await contributions.ExistsAsync(stokvelId, user.Id, request.CycleId))
            throw new DomainConflictException("This member has already contributed for this cycle.");   // CHANGED 6.2: no name

        var contribution = new Contribution(stokvelId, user.Id, request.CycleId, request.Amount);
        await contributions.AddAsync(contribution);
        RondiMetrics.ContributionRecorded();   // NEW 6.2: counted only after the row is really saved

        var response = contribution.ToResponse();
        await idempotencyStore.SaveAsync(idempotencyKey, new IdempotencyRecord(requestHash, response));

        return (response, false);
    }

    private static string ComputeHash(Guid stokvelId, ContributionRequest request)
    {
        var raw = $"{stokvelId}|{request.UserId}|{request.CycleId}|{request.Amount}";
        var bytes = SHA256.HashData(Encoding.UTF8.GetBytes(raw));
        return Convert.ToHexString(bytes);
    }
}

// NEW 8.1: builds "my stokvels" and the stokvel detail. A fixed number of queries, never one per stokvel.
public sealed class MyStokvelsService(RondiTrackDbContext db)
{
    // Every stokvel the user belongs to, with their role in each (empty list if none)
    public async Task<IReadOnlyList<MyStokvelResponse>> ListMineAsync(Guid userId, CancellationToken ct)
    {
        // query 1: my memberships with the stokvel's basic facts
        var memberships = await db.StokvelMembers.AsNoTracking()
            .Where(m => m.UserId == userId)
            .OrderBy(m => m.Stokvel.Name)
            .Select(m => new
            {
                m.StokvelId,
                m.Stokvel.Name,
                m.Stokvel.ContributionAmount,
                m.Stokvel.Frequency,
                m.Role,
                MemberCount = m.Stokvel.Memberships.Count
            })
            .ToListAsync(ct);

        if (memberships.Count == 0) return [];   // no stokvels is still a normal 200 with an empty list

        // queries 2 and 3: current cycle + sums for all those stokvels at once
        var frequencies = memberships.ToDictionary(m => m.StokvelId, m => m.Frequency);
        var cycles = await LoadCurrentCyclesAsync(frequencies, userId, ct);

        return memberships.Select(m =>
        {
            cycles.TryGetValue(m.StokvelId, out var found);
            var summary = cycles.ContainsKey(m.StokvelId) ? found.Summary : null;
            var paid = cycles.ContainsKey(m.StokvelId) ? found.MyPaid : 0m;
            var owed = summary is null ? 0m : Math.Max(m.ContributionAmount - paid, 0m);   // what I still owe this cycle

            return new MyStokvelResponse(
                m.StokvelId, m.Name, m.ContributionAmount, m.Frequency, m.MemberCount, m.Role,
                paid, owed, summary?.NextPayoutDateUtc, summary);
        }).ToList();
    }

    // Detail of ONE stokvel (the controller already checked the caller may see it)
    public async Task<StokvelDetailResponse> GetDetailAsync(Stokvel stokvel, Guid? userId, CancellationToken ct)
    {
        StokvelRole? role = null;
        if (userId is Guid uid)
        {
            role = await db.StokvelMembers.AsNoTracking()
                .Where(m => m.UserId == uid && m.StokvelId == stokvel.Id)
                .Select(m => (StokvelRole?)m.Role)
                .FirstOrDefaultAsync(ct);
        }

        var cycles = await LoadCurrentCyclesAsync(
            new Dictionary<Guid, ContributionFrequency> { [stokvel.Id] = stokvel.Frequency }, userId, ct);
        var summary = cycles.TryGetValue(stokvel.Id, out var found) ? found.Summary : null;

        return new StokvelDetailResponse(
            stokvel.Id, stokvel.Name, stokvel.ContributionAmount, stokvel.Frequency, stokvel.MaxMembers,
            stokvel.Members.Count, stokvel.IsFull, stokvel.PayoutPerCycle, stokvel.CreatedAtUtc,
            role, summary);
    }

    // For each stokvel: its newest OPEN cycle, how much was collected, and how much THIS user paid
    private async Task<Dictionary<Guid, (CurrentCycleSummaryResponse Summary, decimal MyPaid)>> LoadCurrentCyclesAsync(
        Dictionary<Guid, ContributionFrequency> frequencies, Guid? userId, CancellationToken ct)
    {
        var stokvelIds = frequencies.Keys.ToList();

        var open = await db.ContributionCycles.AsNoTracking()
            .Where(c => stokvelIds.Contains(c.StokvelId) && c.Status == CycleStatus.Open)
            .Select(c => new { c.Id, c.StokvelId, c.CycleNumber, c.TargetAmount, c.CreatedAtUtc })
            .ToListAsync(ct);

        // newest open cycle per stokvel
        var latest = open.GroupBy(c => c.StokvelId)
            .Select(g => g.OrderByDescending(c => c.CycleNumber).First())
            .ToList();

        var result = new Dictionary<Guid, (CurrentCycleSummaryResponse, decimal)>();
        if (latest.Count == 0) return result;

        var cycleIds = latest.Select(c => c.Id).ToList();
        var me = userId ?? Guid.Empty;

        // one grouped query: total collected and my share, per cycle (runs in PostgreSQL)
        var sums = (await db.Contributions.AsNoTracking()
            .Where(c => cycleIds.Contains(c.CycleId))
            .GroupBy(c => c.CycleId)
            .Select(g => new
            {
                CycleId = g.Key,
                Total = g.Sum(x => x.Amount),
                Mine = g.Sum(x => x.UserId == me ? x.Amount : 0m)
            })
            .ToListAsync(ct)).ToDictionary(s => s.CycleId);

        foreach (var c in latest)
        {
            sums.TryGetValue(c.Id, out var s);
            var collected = s?.Total ?? 0m;
            var mine = s?.Mine ?? 0m;
            var next = NextPayout(c.CreatedAtUtc, frequencies[c.StokvelId]);
            result[c.StokvelId] = (
                new CurrentCycleSummaryResponse(c.Id, c.CycleNumber, c.TargetAmount, collected, c.CreatedAtUtc, next),
                mine);
        }

        return result;
    }

    // The model has no payout schedule, so the next payout is the cycle start plus one period
    private static DateTime NextPayout(DateTime start, ContributionFrequency frequency) => frequency switch
    {
        ContributionFrequency.Weekly => start.AddDays(7),
        ContributionFrequency.Fortnightly => start.AddDays(14),
        _ => start.AddMonths(1)
    };
}

public sealed class HomeService(RondiTrackDbContext db, TimeProvider time) : IHomeService
{
    public async Task<HomeResponse> GetAsync(Guid userId, CancellationToken ct)
    {
        var now = time.GetUtcNow();   // never DateTime.Now: tests can fake it

        // 1. every stokvel this user is in, with THEIR role in each
        var mine = await db.StokvelMembers.AsNoTracking()
            .Where(m => m.UserId == userId)
            .Select(m => new { m.StokvelId, m.Role, m.Stokvel.Name, m.Stokvel.ContributionAmount, m.Stokvel.Frequency })
            .OrderBy(x => x.Name)
            .ToListAsync(ct);

        if (mine.Count == 0) return new HomeResponse([]);   // empty list, never 404

        var ids = mine.Select(x => x.StokvelId).ToList();

        // 2. the open cycles. CHANGED 7.2: a cycle the job already closed is not "current"
        var openCycles = await db.ContributionCycles.AsNoTracking()
            .Where(c => ids.Contains(c.StokvelId) && c.Status == Domain.CycleStatus.Open && c.ClosedAtUtc == null)
            .OrderByDescending(c => c.CycleNumber)
            .Select(c => new { c.Id, c.StokvelId, c.CycleNumber, c.CreatedAtUtc, c.EndsOn })
            .ToListAsync(ct);

        var cycleIds = openCycles.Select(c => c.Id).ToList();
        // only the caller's own contributions: other members' amounts never leave the server
        var paid = await db.Contributions.AsNoTracking()
            .Where(c => c.UserId == userId && cycleIds.Contains(c.CycleId))
            .Select(c => new { c.CycleId, c.Amount })
            .ToListAsync(ct);

        // NEW 7.2: the caller's OWN arrears per stokvel, added up in cents by the database
        var arrears = await db.Arrears.AsNoTracking()
            .Where(a => a.UserId == userId && ids.Contains(a.StokvelId))
            .GroupBy(a => a.StokvelId)
            .Select(g => new { StokvelId = g.Key, Cents = g.Sum(a => a.ShortfallCents) })
            .ToListAsync(ct);
        var arrearsByStokvel = arrears.ToDictionary(a => a.StokvelId, a => a.Cents);

        var items = mine.Select(s =>
        {
            long? arrearsCents = arrearsByStokvel.TryGetValue(s.StokvelId, out var owedBefore) && owedBefore > 0 ? owedBefore : null;

            var cycle = openCycles.FirstOrDefault(c => c.StokvelId == s.StokvelId);   // newest open one
            if (cycle is null)
                return new HomeStokvelDto(s.StokvelId, s.Name, s.Role, null, 0, 0, null, arrearsCents);

            var start = DateTime.SpecifyKind(cycle.CreatedAtUtc, DateTimeKind.Utc);
            var due = CycleRules.DueAtUtc(cycle.EndsOn, cycle.CreatedAtUtc, s.Frequency);   // CHANGED 7.2: the real due date

            // next payout: unchanged rule, always in the future
            var payout = CycleRules.Advance(new DateTimeOffset(start), s.Frequency);
            while (payout <= now) payout = CycleRules.Advance(payout, s.Frequency);

            // money as integer cents
            var paidCents = Money.ToCents(paid.Where(p => p.CycleId == cycle.Id).Sum(p => p.Amount));
            var dueCents = Money.ToCents(s.ContributionAmount);

            return new HomeStokvelDto(
                s.StokvelId, s.Name, s.Role,
                new HomeCycleDto(cycle.Id, cycle.CycleNumber, Domain.CycleStatus.Open, start, due),
                paidCents, Math.Max(0, dueCents - paidCents), payout.UtcDateTime, arrearsCents);
        }).ToList();

        return new HomeResponse(items);
    }

        await stokvels.UpdateAsync(stokvel);

    public const int MinMembers = 2;
    public const int MaxMembersLimit = 50;
    public const decimal MaxContribution = 1_000_000m;

    private readonly List<User> _members = [];

    public string Name { get; private set; }
    public decimal ContributionAmount { get; private set; }
    public ContributionFrequency Frequency { get; private set; }
    public int MaxMembers { get; private set; }

    // ADDED 5.3: PostgreSQL's built-in row version (xmin). Not a real column.
    public uint Version { get; private set; }

    public IReadOnlyCollection<User> Members => _members.AsReadOnly();
    public decimal PayoutPerCycle => ContributionAmount * _members.Count;
    public bool IsFull => _members.Count >= MaxMembers;

    // The real EF Core navigations (Members above stays for the business rules)
    public List<StokvelMember> Memberships { get; } = [];
    public List<ContributionCycle> Cycles { get; } = [];

    private Stokvel() { Name = null!; } // for EF Core only

    public Stokvel(string name, decimal contributionAmount, ContributionFrequency frequency, int maxMembers)
    {
        Id = Guid.NewGuid();
        CreatedAtUtc = DateTime.UtcNow;
        Name = ValidateName(name);
        ContributionAmount = ValidateAmount(contributionAmount);
        Frequency = ValidateFrequency(frequency);
        MaxMembers = ValidateMaxMembers(maxMembers);
    }

    // CHANGED 5.4: internal -> public, because SeedData now lives in the Infrastructure project
    public Stokvel(Guid id, string name, decimal contributionAmount,
        ContributionFrequency frequency, int maxMembers)
        : this(name, contributionAmount, frequency, maxMembers)
    {
        Id = id;
    }

    // CHANGED 5.4: internal -> public, because EfStokvelRepository now lives in the Infrastructure project
    public void LoadMembers(IEnumerable<User> users)
    {
        _members.Clear();
        _members.AddRange(users);
    }

    // ADDED 5.3: "the client last saw THIS version". Called before saving an update from a PUT.
    public void SetExpectedVersion(uint version) => Version = version;

    public void UpdateDetails(string name, decimal contributionAmount,
        ContributionFrequency frequency, int maxMembers)
    {
        var validName = ValidateName(name);
        var validAmount = ValidateAmount(contributionAmount);
        var validFrequency = ValidateFrequency(frequency);
        var validMax = ValidateMaxMembers(maxMembers);

        if (validMax < _members.Count)
            throw new DomainConflictException(
                $"Cannot reduce capacity to {validMax}: this stokvel already has {_members.Count} members.");   // CHANGED 6.2: no stokvel name

        Name = validName;
        ContributionAmount = validAmount;
        Frequency = validFrequency;
        MaxMembers = validMax;
    }

    public void AddMember(User user)
    {
        ArgumentNullException.ThrowIfNull(user);

        if (HasMember(user.Id))
            throw new DomainConflictException("That user is already a member of this stokvel.");   // CHANGED 6.2: no name

        if (IsFull)
            throw new DomainConflictException($"This stokvel is full ({MaxMembers} members).");   // CHANGED 6.2: no stokvel name

        _members.Add(user);
    }

    public bool RemoveMember(Guid userId) => _members.RemoveAll(m => m.Id == userId) > 0;

    public bool HasMember(Guid userId) => _members.Any(m => m.Id == userId);

    public User? FindMember(Guid userId) => _members.FirstOrDefault(m => m.Id == userId);

    private static string ValidateName(string? name)
    {
        if (string.IsNullOrWhiteSpace(name))
            throw new DomainValidationException("Stokvel name is required.");

        var trimmed = name.Trim();
        if (trimmed.Length > MaxNameLength)
            throw new DomainValidationException($"Stokvel name cannot exceed {MaxNameLength} characters.");

        return trimmed;
    }

    private static decimal ValidateAmount(decimal amount)
    {
        if (amount <= 0)
            throw new DomainValidationException("Contribution amount must be greater than zero.");

        if (amount > MaxContribution)
            throw new DomainValidationException($"Contribution amount cannot exceed R{MaxContribution:N0}.");

        if (decimal.Round(amount, 2) != amount)
            throw new DomainValidationException("Contribution amount cannot have more than 2 decimal places.");

        return amount;
    }

    private static ContributionFrequency ValidateFrequency(ContributionFrequency frequency)
    {
        if (!Enum.IsDefined(frequency))
            throw new DomainValidationException("Frequency must be Weekly, Fortnightly or Monthly.");

        return frequency;
    }

    private static int ValidateMaxMembers(int maxMembers)
    {
        if (maxMembers < MinMembers || maxMembers > MaxMembersLimit)
            throw new DomainValidationException(
                $"Max members must be between {MinMembers} and {MaxMembersLimit}.");

        return maxMembers;
    }
}

            var page = (await _client.GetFromJsonAsync<PageDto>(query))!;
            seen.AddRange(page.Items.Select(i => i.Id));
            token = page.NextPageToken;
        } while (token != "");

        Assert.Equal(3, seen.Count);
        Assert.Equal(3, seen.Distinct().Count());
    }

    {

        Assert.Equal(HttpStatusCode.BadRequest, response.StatusCode);
    }

    private sealed record PageDto(List<IdDto> Items, string NextPageToken);

    [Fact]
    public async Task Rejected_request_does_not_use_its_idempotency_key()
    {
        var userId = await TestSeed.UserAsync(factory.Services);
        var (stokvelId, cycles) = await TestSeed.StokvelAsync(factory.Services, [(userId, StokvelRole.Member)], cycles: 2);

        // 1 token, refills 1 every 2 seconds
        using var f = With(("RateLimiting:Contribution:TokenLimit", "1"),
                           ("RateLimiting:Contribution:TokensPerPeriod", "1"),
                           ("RateLimiting:Contribution:ReplenishSeconds", "2"));
        var client = AuthedClient(f, userId);

        // 1) uses the only token (cycle 1)
        Assert.Equal(HttpStatusCode.Created, (await client.SendAsync(ContributionPost(stokvelId, userId, cycles[0], "key-first"))).StatusCode);

        // 2) key K is rejected with 429 (must NOT be stored)
        var k = $"key-{Guid.NewGuid():N}";
        Assert.Equal(HttpStatusCode.TooManyRequests, (await client.SendAsync(ContributionPost(stokvelId, userId, cycles[1], k))).StatusCode);

        await Task.Delay(TimeSpan.FromSeconds(2.5));   // wait out the short window

        // 3) same K now succeeds exactly once
        var ok = await client.SendAsync(ContributionPost(stokvelId, userId, cycles[1], k));
        Assert.Equal(HttpStatusCode.Created, ok.StatusCode);
        Assert.False(ok.Headers.Contains("Idempotency-Replayed"));

        // 4) third time = replay, not a second contribution
        await Task.Delay(TimeSpan.FromSeconds(2.5));   // let the bucket refill so it reaches the idempotency check
        var replay = await client.SendAsync(ContributionPost(stokvelId, userId, cycles[1], k));
        Assert.Equal(HttpStatusCode.Created, replay.StatusCode);
        Assert.Equal("true", replay.Headers.GetValues("Idempotency-Replayed").Single());

        // 5) exactly one row for that cycle
        using var scope = factory.Services.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<RondiTrackDbContext>();
        Assert.Equal(1, await db.Contributions.CountAsync(c => c.CycleId == cycles[1] && c.UserId == userId));
    }
