using Microsoft.AspNetCore.Identity;
using RondiTrack.Dtos;

namespace RondiTrack.Auth;

// What the controller needs after login or refresh
public sealed record AuthResult(
    string AccessToken,
    DateTime AccessTokenExpiresAtUtc,
    string RefreshToken,              // goes ONLY into the cookie
    DateTime RefreshTokenExpiresAtUtc,
    Guid RefreshTokenId);

public interface IAuthService
{
    Task<User> RegisterAsync(RegisterRequest request);
    Task<AuthResult> LoginAsync(LoginRequest request);
    Task<AuthResult> RefreshAsync(string? rawRefreshToken);
    Task LogoutAsync(string? rawRefreshToken);
}

public sealed class AuthService(
    IUserRepository users,
    IRefreshTokenRepository tokens,
    IPasswordHasher<User> hasher,
    ITokenService tokenService,
    JwtOptions jwt,
    ILogger<AuthService> logger) : IAuthService
{
    // A fake user and hash. Used when the email is unknown, so the time taken looks the same as a wrong password.
    private static readonly User DummyUser = new("Dummy", "dummy@example.com");
    private static readonly string DummyHash =
        new PasswordHasher<User>().HashPassword(DummyUser, "dummy-password-never-used");

    public async Task<User> RegisterAsync(RegisterRequest request)
    {
        var user = new User(request.FullName, request.Email);   // trims + lower-cases the email

        if (await users.EmailExistsAsync(user.Email))
            throw new DomainConflictException("An account with this email already exists.");   // 409

        user.SetPasswordHash(hasher.HashPassword(user, request.Password));   // store the hash only
        await users.AddAsync(user);   // the unique index also catches two requests at the same moment

        logger.LogInformation("User {UserId} registered", user.Id);   // id only, never email or password
        return user;
    }

    public async Task<AuthResult> LoginAsync(LoginRequest request)
    {
        var user = await users.GetByEmailAsync(request.Email);

        // always run one password check (real or fake hash) so unknown email and wrong password look alike
        var hash = user?.PasswordHash ?? DummyHash;
        var outcome = hasher.VerifyHashedPassword(user ?? DummyUser, hash, request.Password);

        if (user is null || user.PasswordHash is null || outcome == PasswordVerificationResult.Failed)
        {
            // CHANGED 6.2: never the typed email. Our own user id when the account exists, so a guessing attack on one account is visible.
            if (user is null)
                logger.LogWarning("Login failed: no matching account");
            else
                logger.LogWarning("Login failed for user {UserId}", user.Id);
            throw new InvalidCredentialsException();   // same 401 message for both cases
        }

        var result = await IssueAsync(user, Guid.NewGuid());   // new login = new token family
        logger.LogInformation("Login succeeded for user {UserId}", user.Id);
        return result;
    }

    public async Task<AuthResult> RefreshAsync(string? rawRefreshToken)
    {
        if (string.IsNullOrWhiteSpace(rawRefreshToken))
            throw new InvalidRefreshTokenException();

        var now = DateTime.UtcNow;
        var stored = await tokens.GetByHashAsync(tokenService.HashRefreshToken(rawRefreshToken))
            ?? throw new InvalidRefreshTokenException();

        // REUSE DETECTION: this token was already used (or revoked) -> someone replayed it
        if (stored.UsedAtUtc.HasValue || stored.RevokedAtUtc.HasValue)
        {
            await tokens.RevokeFamilyAsync(stored.FamilyId, now);   // kill every token of this login
            logger.LogWarning("Refresh token reuse detected for user {UserId}. Session {FamilyId} revoked",
                stored.UserId, stored.FamilyId);
            throw new InvalidRefreshTokenException();
        }

        if (stored.ExpiresAtUtc <= now)
            throw new InvalidRefreshTokenException();

        // read the user again, so the new access token reflects their CURRENT platform flag
        var user = await users.GetByIdAsync(stored.UserId)
            ?? throw new InvalidRefreshTokenException();

        var result = await IssueAsync(user, stored.FamilyId);   // ROTATION: new token, same family

        // mark the old one used. If another request got there first, this is a replay.
        if (!await tokens.TryMarkUsedAsync(stored.Id, result.RefreshTokenId, now))
        {
            await tokens.RevokeFamilyAsync(stored.FamilyId, now);
            logger.LogWarning("Refresh token reuse detected for user {UserId}. Session {FamilyId} revoked",
                stored.UserId, stored.FamilyId);
            throw new InvalidRefreshTokenException();
        }

        logger.LogInformation("Refresh token rotated for user {UserId}", user.Id);
        return result;
    }

    // Idempotent: unknown or missing token is fine, it just does nothing
    public async Task LogoutAsync(string? rawRefreshToken)
    {
        if (string.IsNullOrWhiteSpace(rawRefreshToken)) return;

        var stored = await tokens.GetByHashAsync(tokenService.HashRefreshToken(rawRefreshToken));
        if (stored is null) return;

        await tokens.RevokeFamilyAsync(stored.FamilyId, DateTime.UtcNow);
        logger.LogInformation("User {UserId} logged out", stored.UserId);
    }

    // Makes the access token + a new refresh token (stored as a hash) for one family
    private async Task<AuthResult> IssueAsync(User user, Guid familyId)
    {
        var now = DateTime.UtcNow;
        var access = tokenService.CreateAccessToken(user.Id, user.IsPlatformAdmin);
        var (raw, hash) = tokenService.CreateRefreshToken();

        var refresh = new RefreshToken(user.Id, familyId, hash, now, now.AddDays(jwt.RefreshTokenDays));
        await tokens.AddAsync(refresh);

        return new AuthResult(access.Value, access.ExpiresAtUtc, raw, refresh.ExpiresAtUtc, refresh.Id);
    }
}

using System.Security.Claims;
using System.Security.Cryptography;
using System.Text;
using Microsoft.AspNetCore.WebUtilities;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace RondiTrack.Auth;

public sealed record AccessToken(string Value, DateTime ExpiresAtUtc);

public interface ITokenService
{
    AccessToken CreateAccessToken(Guid userId, bool isPlatformAdmin);
    (string Raw, string Hash) CreateRefreshToken();
    string HashRefreshToken(string raw);
}

public sealed class TokenService(JwtOptions options) : ITokenService
{
    // the signing key as bytes (HS256 needs at least 32)
    private readonly SymmetricSecurityKey _key = new(Encoding.UTF8.GetBytes(options.SigningKey));

    public AccessToken CreateAccessToken(Guid userId, bool isPlatformAdmin)
    {
        var now = DateTime.UtcNow;
        var expires = now.AddMinutes(options.AccessTokenMinutes);

        // Only facts that can safely be stale for a few minutes. NO stokvel roles here.
        var claims = new List<Claim>
        {
            new("sub", userId.ToString()),          // who the user is (never changes)
            new("jti", Guid.NewGuid().ToString())   // unique id of this token
        };
        if (isPlatformAdmin)
            claims.Add(new Claim("role", AuthConstants.PlatformAdminRole));   // platform-level only

        var descriptor = new SecurityTokenDescriptor
        {
            Issuer = options.Issuer,
            Audience = options.Audience,
            Subject = new ClaimsIdentity(claims),
            IssuedAt = now,
            NotBefore = now,
            Expires = expires,
            SigningCredentials = new SigningCredentials(_key, SecurityAlgorithms.HmacSha256)
        };

        return new AccessToken(new JsonWebTokenHandler().CreateToken(descriptor), expires);
    }

    // Random bytes, NOT a JWT. The raw value goes to the client; only its hash is stored.
    public (string Raw, string Hash) CreateRefreshToken()
    {
        var raw = WebEncoders.Base64UrlEncode(RandomNumberGenerator.GetBytes(32));
        return (raw, HashRefreshToken(raw));
    }

    // SHA-256 is enough here because the token is 256 bits of randomness (not a guessable password)
    public string HashRefreshToken(string raw) =>
        Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(raw)));
}

using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using RondiTrack.Telemetry;

public interface IStokvelAccess
{
    Task EnsureAsync(ClaimsPrincipal user, Guid stokvelId, string policy, Guid? onBehalfOfUserId = null);
}

// One place that decides 404 vs 403. No try/catch, no role names.
public sealed class StokvelAccess(IAuthorizationService authorization, ILogger<StokvelAccess> logger) : IStokvelAccess   // NEW 6.2: logger
{
    public async Task EnsureAsync(ClaimsPrincipal user, Guid stokvelId, string policy, Guid? onBehalfOfUserId = null)
    {
        var result = await authorization.AuthorizeAsync(user, new StokvelResource(stokvelId, onBehalfOfUserId), policy);
        if (result.Succeeded) return;

        var notMember = result.Failure?.FailureReasons
            .Any(r => r.Message == StokvelAccessHandler.NotMemberReason) ?? false;

        // not a member: pretend the stokvel does not exist (same text as the real 404)
        if (notMember)
        {
            RondiMetrics.AuthorizationRefused(RefusalResults.NotFound);   // NEW 6.2: count it
            // NEW 6.2: Information. A refusal is an expected answer; a spike is caught by the metric. Our user id only.
            logger.LogInformation("Stokvel access refused ({Result}) for user {UserId}",
                RefusalResults.NotFound, user.GetUserId());
            throw new DomainNotFoundException($"Stokvel {stokvelId} was not found.");
        }

        // a member without the role already knows the stokvel exists
        RondiMetrics.AuthorizationRefused(RefusalResults.Forbidden);      // NEW 6.2: count it
        logger.LogInformation("Stokvel access refused ({Result}) for user {UserId}",
            RefusalResults.Forbidden, user.GetUserId());
        throw new ForbiddenException("You do not have the required role in this stokvel.");
    }
}

using Microsoft.AspNetCore.Authorization;

namespace RondiTrack.Auth;

// What level of access the endpoint needs
public enum StokvelAccessLevel
{
    Member,              // any member
    Officer,             // Treasurer or Admin of THAT stokvel
    RecordContribution   // own contribution, or officer for someone else's
}

// The custom requirement
public sealed record StokvelAccessRequirement(StokvelAccessLevel Level) : IAuthorizationRequirement;

// The resource being protected: one stokvel (and optionally who the action is for)
public sealed record StokvelResource(Guid StokvelId, Guid? OnBehalfOfUserId = null);

// Resource-based handler. It asks the DATABASE every time, so a removed or demoted member is refused at once.
public sealed class StokvelAccessHandler(IStokvelMemberRepository members)
    : AuthorizationHandler<StokvelAccessRequirement, StokvelResource>
{
    public const string NotMemberReason = "not-a-member";     // becomes 404
    public const string MissingRoleReason = "missing-role";   // becomes 403

    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext context, StokvelAccessRequirement requirement, StokvelResource resource)
    {
        // platform admins (support staff) pass on every stokvel
        if (context.User.IsPlatformAdmin())
        {
            context.Succeed(requirement);
            return;
        }

        var callerId = context.User.GetUserId();
        if (callerId is null) return;   // not signed in: no success

        // the real check: the caller's membership row for THIS stokvel
        var membership = await members.GetAsync(callerId.Value, resource.StokvelId);
        if (membership is null)
        {
            context.Fail(new AuthorizationFailureReason(this, NotMemberReason));
            return;
        }

        var isOfficer = membership.Role is StokvelRole.Treasurer or StokvelRole.Admin;

        var needsOfficer = requirement.Level switch
        {
            StokvelAccessLevel.Officer => true,
            // recording for yourself is fine; for someone else you must be an officer
            StokvelAccessLevel.RecordContribution =>
                resource.OnBehalfOfUserId is Guid target && target != callerId.Value,
            _ => false
        };

        if (needsOfficer && !isOfficer)
        {
            context.Fail(new AuthorizationFailureReason(this, MissingRoleReason));
            return;
        }

        context.Succeed(requirement);
    }
}

[Route($"{ApiRoutes.V1}/auth")]
[AllowAnonymous]   // all four endpoints work without an access token
public class AuthController(IAuthService auth) : RondiControllerBase
{
    /// <summary>Registers a new account. Password must be at least 12 characters.</summary>
    [HttpPost("register")]
    [ProducesResponseType(typeof(UserResponse), StatusCodes.Status201Created)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status400BadRequest)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status409Conflict)]
    public async Task<ActionResult<UserResponse>> Register(RegisterRequest request)
    {
        var user = await auth.RegisterAsync(request);
        return Created($"{ApiRoutes.V1Prefix}/users/{user.Id}", user.ToResponse());
    }

    /// <summary>Signs in. Returns the access token in the body and sets the refresh token as a cookie.</summary>
    [HttpPost("login")]
    [EnableRateLimiting(RateLimitPolicies.Login)]
    [ProducesResponseType(typeof(AuthResponse), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status401Unauthorized)]
    public async Task<ActionResult<AuthResponse>> Login(LoginRequest request) =>
        TokenResponse(await auth.LoginAsync(request));

    /// <summary>Swaps the refresh cookie for a new access token and a new refresh cookie (rotation).</summary>
    [HttpPost("refresh")]
    [ProducesResponseType(typeof(AuthResponse), StatusCodes.Status200OK)]
    [ProducesResponseType(typeof(ProblemDetails), StatusCodes.Status401Unauthorized)]
    public async Task<ActionResult<AuthResponse>> Refresh()
    {
        Request.Cookies.TryGetValue(AuthConstants.RefreshCookieName, out var raw);   // cookie is the only source
        return TokenResponse(await auth.RefreshAsync(raw));
    }

    /// <summary>Signs out: revokes the session's refresh tokens. Safe to call more than once.</summary>
    [HttpPost("logout")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
    public async Task<IActionResult> Logout()
    {
        Request.Cookies.TryGetValue(AuthConstants.RefreshCookieName, out var raw);
        await auth.LogoutAsync(raw);   // does nothing if there is no valid cookie
        Response.Cookies.Delete(AuthConstants.RefreshCookieName, CookieOptionsFor(null));   // same path as when it was set
        return NoContent();
    }

    // Sets the refresh cookie and returns ONLY the access token in the body
    private ActionResult<AuthResponse> TokenResponse(AuthResult result)
    {
        Response.Cookies.Append(AuthConstants.RefreshCookieName, result.RefreshToken,
            CookieOptionsFor(result.RefreshTokenExpiresAtUtc));
        Response.Headers.CacheControl = "no-store";   // tokens must never be cached
        return Ok(new AuthResponse(result.AccessToken, result.AccessTokenExpiresAtUtc, "Bearer"));
    }

    // The cookie flags (each one is justified in the README)
    private static CookieOptions CookieOptionsFor(DateTime? expiresUtc) => new()
    {
        HttpOnly = true,                              // JavaScript cannot read it, so XSS cannot steal it
        Secure = true,                                // only sent over HTTPS
        SameSite = SameSiteMode.Strict,               // never sent on cross-site requests (CSRF)
        Path = AuthConstants.RefreshCookiePath,       // only sent to /api/auth/*
        IsEssential = true,                           // not blocked by consent middleware
        Expires = expiresUtc.HasValue ? new DateTimeOffset(expiresUtc.Value) : null   // lives as long as the token
    };
}

namespace RondiTrack.Errors;

public sealed class RondiExceptionHandler(ILogger<RondiExceptionHandler> logger) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(HttpContext httpContext, Exception exception, CancellationToken cancellationToken)
    {
        // UseExceptionHandler wipes response headers (including CORS), so put them back
        await CorsSetup.ReapplyAsync(httpContext);

        var correlationId = CorrelationId.For(httpContext);

        var status = StatusCodes.Status500InternalServerError;
        var title = "Internal Server Error";
        var detail = "An unexpected error occurred.";
        Dictionary<string, string[]>? errors = null;   // field -> messages, validation only

        if (exception is ValidationException vex)
        {
            status = StatusCodes.Status400BadRequest;
            title = "Validation Failed";
            detail = "One or more fields are invalid.";
            errors = vex.Errors
                .GroupBy(e => JsonNamingPolicy.CamelCase.ConvertName(e.PropertyName))
                .ToDictionary(g => g.Key, g => g.Select(e => e.ErrorMessage).ToArray());
        }
        else if (exception is DomainNotFoundException) { status = 404; title = "Not Found"; detail = exception.Message; }
        else if (exception is DomainConflictException) { status = 409; title = "Conflict"; detail = exception.Message; }
        else if (exception is DomainReferenceException) { status = 422; title = "Unprocessable Entity"; detail = exception.Message; }
        else if (exception is DomainValidationException) { status = 400; title = "Bad Request"; detail = exception.Message; }
        else if (exception is ApiBadRequestException) { status = 400; title = "Bad Request"; detail = exception.Message; }
        else if (exception is PreconditionRequiredException) { status = 428; title = "Precondition Required"; detail = exception.Message; }
        else if (exception is InvalidCredentialsException or InvalidRefreshTokenException)
        {
            status = 401; title = "Unauthorized"; detail = exception.Message;
            httpContext.Response.Headers.WWWAuthenticate = "Bearer";
        }
        else if (exception is ForbiddenException) { status = 403; title = "Forbidden"; detail = exception.Message; }
        else if (exception is DbUpdateConcurrencyException)
        {
            if (httpContext.Request.Headers.ContainsKey("If-Match"))
            {
                status = 412; title = "Precondition Failed";
                detail = "This record was changed by someone else since you loaded it. Reload it and try again.";
            }
            else
            {
                status = 409; title = "Conflict";
                detail = "This record was changed by someone else at the same time. Reload it and try again.";
            }
        }
        else if (exception is DbUpdateException { InnerException: PostgresException { SqlState: PostgresErrorCodes.UniqueViolation } })
        {
            status = 409; title = "Conflict";
            detail = "That record already exists (it would break a uniqueness rule).";
        }

        if (status >= 500)
            logger.LogError(exception, "Request failed with status {Status}. CorrelationId: {CorrelationId}", status, correlationId);
        else
            logger.LogInformation("Request rejected with status {Status} ({ExceptionType}). CorrelationId: {CorrelationId}",
                status, exception.GetType().Name, correlationId);

        var problemDetails = new ProblemDetails { Status = status, Title = title, Detail = detail };
        problemDetails.Extensions["correlationId"] = correlationId;
        if (errors is not null) problemDetails.Extensions["errors"] = errors;

        httpContext.Response.StatusCode = status;
        httpContext.Response.ContentType = "application/problem+json";
        await httpContext.Response.WriteAsync(JsonSerializer.Serialize(problemDetails), cancellationToken);
        return true;
    }
}

using Microsoft.AspNetCore.Cors.Infrastructure;

namespace RondiTrack.Web;

public static class CorsSetup
{
    public static IServiceCollection AddFrontendCors(this IServiceCollection services)
    {
        services.AddCors();
        // Options pattern = read lazily, so test UseSetting overrides are always seen
        services.AddOptions<CorsOptions>().Configure<IConfiguration>((o, config) =>
        {
            var origins = config.GetSection("Cors:AllowedOrigins").Get<string[]>() ?? [];
            if (origins.Any(x => x.Contains('*') || x.EndsWith('/')))
                throw new InvalidOperationException("Cors:AllowedOrigins must be exact origins: no '*', no trailing slash.");

            o.AddPolicy(CorsPolicies.Frontend, p => p
                .WithOrigins(origins)                                   // empty list = nobody allowed
                .WithMethods("GET", "POST", "PUT", "PATCH", "DELETE")
                .WithHeaders("Content-Type", "Authorization", ApiHeaders.IdempotencyKey, "If-Match", "If-None-Match")
                .WithExposedHeaders("Retry-After", ApiHeaders.IdempotencyReplayed, "ETag", "Location")
                .AllowCredentials()
                .SetPreflightMaxAge(TimeSpan.FromMinutes(10)));
        });
        return services;
    }

    /// The exception handler clears response headers, so CORS headers set earlier are lost on
    /// exception-based errors (404/403/409/500). Call this at the top of the handler to put them back.
    public static async Task ReapplyAsync(HttpContext ctx)
    {
        var sp = ctx.RequestServices;
        if (sp is null) return;                                   // bare DefaultHttpContext in unit tests: nothing to reapply
        var provider = sp.GetService<ICorsPolicyProvider>();
        var service = sp.GetService<ICorsService>();
        if (provider is null || service is null) return;          // CORS not registered
        var policy = await provider.GetPolicyAsync(ctx, CorsPolicies.Frontend);
        if (policy is null) return;
        service.ApplyResult(service.EvaluatePolicy(ctx, policy), ctx.Response);
    }
}

using System.Security.Claims;
using System.Threading.RateLimiting;
using Microsoft.AspNetCore.RateLimiting;
using RondiTrack.Telemetry;

namespace RondiTrack.Web;

public static class RateLimitingSetup
{
    public static IServiceCollection AddRondiRateLimiting(this IServiceCollection services)
    {
        services.AddRateLimiter(_ => { });
        services.AddOptions<RateLimiterOptions>().Configure<IConfiguration>((o, cfg) =>
        {
            o.RejectionStatusCode = StatusCodes.Status429TooManyRequests;   // default is 503
            o.OnRejected = WriteRejectionAsync;

            var login = cfg.GetSection("RateLimiting:Login");
            o.AddPolicy(RateLimitPolicies.Login, ctx =>
                RateLimitPartition.GetFixedWindowLimiter(ClientIp(ctx), _ => new FixedWindowRateLimiterOptions
                {
                    PermitLimit = login.GetValue("PermitLimit", 5),
                    Window = TimeSpan.FromSeconds(login.GetValue("WindowSeconds", 60.0)),
                    QueueLimit = 0,
                    AutoReplenishment = true
                }));

            o.AddPolicy(RateLimitPolicies.Contribution, ctx =>
                RateLimitPartition.GetTokenBucketLimiter(UserKey(ctx), _ => Bucket(cfg.GetSection("RateLimiting:Contribution"))));

            o.AddPolicy(RateLimitPolicies.Payout, ctx =>
                RateLimitPartition.GetTokenBucketLimiter(UserKey(ctx), _ => Bucket(cfg.GetSection("RateLimiting:Payout"))));
        });
        return services;
    }

    private static TokenBucketRateLimiterOptions Bucket(IConfigurationSection s) => new()
    {
        TokenLimit = s.GetValue("TokenLimit", 30),
        TokensPerPeriod = s.GetValue("TokensPerPeriod", 5),
        ReplenishmentPeriod = TimeSpan.FromSeconds(s.GetValue("ReplenishSeconds", 60.0)),
        QueueLimit = 0,              // reject, never hold a money request open
        AutoReplenishment = true
    };

    private static string ClientIp(HttpContext c) => "ip:" + (c.Connection.RemoteIpAddress?.ToString() ?? "unknown");

    // signed-in user id; falls back to IP only for unauthenticated callers (they'll get 401 afterwards)
    private static string UserKey(HttpContext c) =>
        c.User.FindFirstValue("sub") is { Length: > 0 } sub ? "user:" + sub : ClientIp(c);

    private static async ValueTask WriteRejectionAsync(OnRejectedContext ctx, CancellationToken ct)
    {
        var http = ctx.HttpContext;
        var seconds = 1;
        if (ctx.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retry))
            seconds = Math.Max(1, (int)Math.Ceiling(retry.TotalSeconds));

        http.Response.StatusCode = StatusCodes.Status429TooManyRequests;
        http.Response.Headers.RetryAfter = seconds.ToString();

        var problem = new Microsoft.AspNetCore.Mvc.ProblemDetails
        {
            Status = 429,
            Title = "Too many requests",
            Detail = $"Rate limit reached. Try again in {seconds} second(s).",
            Type = "https://www.rfc-editor.org/rfc/rfc6585#section-4"
        };
        problem.Extensions["correlationId"] = CorrelationId.For(http);   // your 6.2 helper
        await http.Response.WriteAsJsonAsync(problem, options: null, contentType: "application/problem+json", ct);
    }
}

            throw new DomainValidationException("Full name is required.");

        if (trimmed.Length > MaxNameLength)
            throw new DomainValidationException($"Full name cannot exceed {MaxNameLength} characters.");

namespace RondiTrack.Data;

// NEW 6.1: refresh tokens in PostgreSQL
public sealed class EfRefreshTokenRepository(RondiTrackDbContext db) : IRefreshTokenRepository
{
    public async Task AddAsync(RefreshToken token)
    {
        db.RefreshTokens.Add(token);
        await db.SaveChangesAsync();
    }

    // pure read, so not tracked
    public async Task<RefreshToken?> GetByHashAsync(string tokenHash) =>
        await db.RefreshTokens.AsNoTracking().FirstOrDefaultAsync(t => t.TokenHash == tokenHash);

    // One UPDATE with a WHERE that only matches an unused, unrevoked token.
    // If two requests race, only one gets "1 row changed"; the other gets 0 and is treated as reuse.
    public async Task<bool> TryMarkUsedAsync(Guid tokenId, Guid replacedByTokenId, DateTime nowUtc)
    {
        var rows = await db.RefreshTokens
            .Where(t => t.Id == tokenId && t.UsedAtUtc == null && t.RevokedAtUtc == null)
            .ExecuteUpdateAsync(s => s
                .SetProperty(t => t.UsedAtUtc, (DateTime?)nowUtc)
                .SetProperty(t => t.ReplacedByTokenId, (Guid?)replacedByTokenId));
        return rows == 1;
    }

    // Revokes every not-yet-revoked token of this login (the whole family)
    public async Task RevokeFamilyAsync(Guid familyId, DateTime nowUtc) =>
        await db.RefreshTokens
            .Where(t => t.FamilyId == familyId && t.RevokedAtUtc == null)
            .ExecuteUpdateAsync(s => s.SetProperty(t => t.RevokedAtUtc, (DateTime?)nowUtc));
}

    public static async Task<(Guid StokvelId, Guid[] CycleIds)> StokvelAsync(
        IServiceProvider sp, (Guid UserId, StokvelRole Role)[] members, int cycles = 1,
        decimal amount = 500m, DateOnly? firstEndsOn = null,
        ContributionFrequency freq = ContributionFrequency.Monthly)
    {
        using var scope = sp.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<RondiTrackDbContext>();

        db.Stokvels.Add(s);
        await db.SaveChangesAsync();                         // parent first

        for (var i = 0; i < members.Length; i++)
            db.StokvelMembers.Add(new StokvelMember(s.Id, members[i].UserId, i + 1, members[i].Role));

        var ids = new List<Guid>();

            db.ContributionCycles.Add(c);
            ids.Add(c.Id);
        }
        await db.SaveChangesAsync();
        return (s.Id, ids.ToArray());
    }
